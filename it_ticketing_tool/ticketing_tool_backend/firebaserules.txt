rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper function to check if a user is authenticated
    function isAuthenticated() {
      return request.auth != null;
    }

    // Helper function to get the current user's UID
    function getUserId() {
      return request.auth.uid;
    }

    // Helper: support, admin, or super_admin
    function hasSupportAdminOrSuperAdminRole() {
      return exists(/databases/$(database)/documents/users/$(getUserId())) && (
        get(/databases/$(database)/documents/users/$(getUserId())).data.role == 'support' ||
        get(/databases/$(database)/documents/users/$(getUserId())).data.role == 'admin' ||
        get(/databases/$(database)/documents/users/$(getUserId())).data.role == 'super_admin'
      );
    }

    // NEW HELPER: For Super Admin specific access
    function hasSuperAdminRole() {
      return exists(/databases/$(database)/documents/users/$(getUserId())) &&
             get(/databases/$(database)/documents/users/$(getUserId())).data.role == 'super_admin';
    }

    // Helper: site_admin can read tickets for their own company/client
    function isSiteAdminForTicket() {
      return exists(/databases/$(database)/documents/users/$(getUserId())) &&
        get(/databases/$(database)/documents/users/$(getUserId())).data.role == 'site_admin' &&
        get(/databases/$(database)/documents/users/$(getUserId())).data.client_name == resource.data.client_name;
    }

    match /tickets/{ticketId} {
      allow read: if isAuthenticated() && (
        resource.data.reporter_id == getUserId() ||
        hasSupportAdminOrSuperAdminRole() ||
        isSiteAdminForTicket()
      );

      // Allow authenticated users to create tickets
      allow create: if isAuthenticated() &&
        request.resource.data.reporter_id == getUserId();

      // Allow authenticated users to update tickets
      allow update: if isAuthenticated() && (
        resource.data.reporter_id == getUserId() ||
        hasSupportAdminOrSuperAdminRole()
      );

      // Deny delete by default, adjust if deletion functionality is needed
      allow delete: if false;
    }

    // Rules for the 'users' collection
    match /users/{userId} {
      // Allow users to read their own profile data (e.g., roles)
      allow read: if isAuthenticated() && (
        getUserId() == userId || hasSupportAdminOrSuperAdminRole()
      );
      // Allow users to update their own profile (e.g., mustChangePassword)
      allow update: if isAuthenticated() && getUserId() == userId;
      allow create: if false;
      allow delete: if false;
    }

    // --- NEW RULES FOR 'clients' COLLECTION ---
    match /clients/{clientId} {
      // Allow authenticated users to read client data.
      allow read: if isAuthenticated();
      // Allow only Super Admins to create, update, or delete client documents.
      allow create, update, delete: if hasSuperAdminRole();
    }
    // --- END NEW RULES FOR 'clients' COLLECTION ---

    // Public artifacts (e.g., app-wide configurations if any)
    match /artifacts/{appId}/public/data/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Private user data (if any)
    match /artifacts/{appId}/users/{userId}/{document=**} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
  }
}
